function cetakPerKelasPerMapel(kelas, mapel, siswa) { const dataHasilMapel = hasilUjianGlobal[mapel.id] || {}; // ⬇️ FILTER: hanya siswa yang sudah mengerjakan mapel ini const siswaFiltered = siswa.filter(s => dataHasilMapel[s.nisn || s.kunci]); if (siswaFiltered.length === 0) { alert('Belum ada siswa yang mengerjakan mapel ini di kelas tersebut.'); return; } let html = `

DAFTAR NILAI UJIAN

Mapel: ${escapeHtml(mapel.nama_mapel || '')} · ${escapeHtml(kelas)}${mapel.guru ? ' · Guru: ' + escapeHtml(mapel.guru) : ''}

`; siswaFiltered.forEach((s, i) => { const nisn = s.nisn || s.kunci; const h = dataHasilMapel[nisn] || null; let benarPg = '-', nilaiPg = '-', nilaiEssay = '-', nilaiAkhir = '-', statusTeks = 'Belum Ujian'; if (h) { const benar = h.benar_pg != null ? h.benar_pg : 0; const total = h.total_pg != null ? h.total_pg : 0; benarPg = `${benar}/${total}`; nilaiPg = h.nilai_pg != null ? h.nilai_pg : '-'; nilaiEssay = h.nilai_essay != null ? h.nilai_essay : '-'; nilaiAkhir = hitungNilaiAkhirSiswa(h); if (h.status === 'dinilai_guru') statusTeks = 'Sudah Dinilai'; else if (h.status === 'selesai') statusTeks = 'Selesai'; else if (h.status === 'menunggu_penilaian') statusTeks = 'Menunggu Essay'; else statusTeks = h.status || '-'; } html += ``; }); html += `
NoNama SiswaNISNBenar PGNilai PGNilai EssayNilai AkhirStatus
${i+1}${escapeHtml(s.nama || '-')}${escapeHtml(nisn || '-')}${escapeHtml(benarPg)}${escapeHtml(String(nilaiPg))}${escapeHtml(String(nilaiEssay))}${escapeHtml(String(nilaiAkhir))}${escapeHtml(statusTeks)}

Dicetak: ${new Date().toLocaleString('id-ID')} · Total: ${siswaFiltered.length} siswa

`; tampilkanOverlayCetak(html, mapel.nama_mapel + ' - ' + kelas); }
